cmake_minimum_required(VERSION 3.25)
project(AlternativeFastDelivery VERSION 0.1.0 LANGUAGES CXX)
set(AFD_VERSION "${PROJECT_VERSION}" CACHE STRING "Application version")
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" AFD_NUMERIC_VERSION "${AFD_VERSION}")
if(AFD_NUMERIC_VERSION)
  set(AFD_APP_MAJOR ${CMAKE_MATCH_1})
  set(AFD_APP_MINOR ${CMAKE_MATCH_2})
  set(AFD_APP_PATCH ${CMAKE_MATCH_3})
else()
  set(AFD_NUMERIC_VERSION ${PROJECT_VERSION})
  set(AFD_APP_MAJOR 0)
  set(AFD_APP_MINOR 1)
  set(AFD_APP_PATCH 0)
endif()
if(AFD_APP_MAJOR GREATER 2047 OR AFD_APP_MINOR GREATER 999 OR AFD_APP_PATCH GREATER 999)
  message(FATAL_ERROR "Version exceeds Android version-code component limits")
endif()
math(EXPR AFD_ANDROID_VERSION_CODE "${AFD_APP_MAJOR} * 1000000 + ${AFD_APP_MINOR} * 1000 + ${AFD_APP_PATCH}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(AFD_BUILD_UI "Build native Qt Quick application" ON)
include(CTest)
if(UNIX AND NOT APPLE AND NOT ANDROID)
  set(QT_DEPLOY_USE_PATCHELF ON)
endif()
find_package(OpenSSL 3 REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
find_package(LibtorrentRasterbar 2.1 CONFIG REQUIRED)
find_package(mailio CONFIG REQUIRED)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core)
add_library(afd_core
  src/core/source.cpp src/email/message.cpp src/email/mbox.cpp
  src/dkim/dkim.cpp src/indexing/index.cpp src/torrent/engine.cpp)
target_include_directories(afd_core PUBLIC src)
if(ANDROID)
  target_compile_definitions(afd_core PUBLIC _FILE_OFFSET_BITS=64)
endif()
if(WIN32)
  target_compile_definitions(afd_core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX)
endif()
target_link_libraries(afd_core PUBLIC OpenSSL::Crypto unofficial::sqlite3::sqlite3
  LibtorrentRasterbar::torrent-rasterbar mailio::mailio Qt6::Core)
if(MSVC)
  target_compile_options(afd_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(afd_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AFD_BUILD_UI)
  find_package(Qt6 6.10 REQUIRED COMPONENTS Quick QuickControls2 Network Concurrent Svg)
  qt_standard_project_setup(REQUIRES 6.8)
  add_subdirectory(src/ui)
endif()
install(FILES README.md DESTINATION share/alternativefastdelivery)
install(DIRECTORY docs DESTINATION share/alternativefastdelivery OPTIONAL)
if(BUILD_TESTING AND NOT ANDROID AND NOT IOS)
  add_executable(afd_scan_benchmark tests/scan_benchmark.cpp)
  target_link_libraries(afd_scan_benchmark PRIVATE afd_core)
  add_executable(afd_tests tests/core_tests.cpp)
  target_link_libraries(afd_tests PRIVATE afd_core)
  add_test(NAME core COMMAND afd_tests)
  add_executable(afd_torrent_test tests/torrent_test.cpp)
  target_link_libraries(afd_torrent_test PRIVATE afd_core)
  add_test(NAME selective_loopback COMMAND afd_torrent_test)
  set_tests_properties(selective_loopback PROPERTIES TIMEOUT 90)
  if(AFD_BUILD_UI)
    find_package(Qt6 6.10 REQUIRED COMPONENTS Test)
    add_executable(afd_ui_test tests/ui_test.cpp src/ui/controller.cpp src/ui/controller.hpp)
    target_link_libraries(afd_ui_test PRIVATE afd_core Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent Qt6::Test)
    target_compile_definitions(afd_ui_test PRIVATE AFD_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    qt_add_resources(afd_ui_test "test_brand" PREFIX "/qt/qml/AFD" FILES resources/afd.svg)
    add_test(NAME native_workflow COMMAND afd_ui_test)
    set_tests_properties(native_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60)
    add_test(NAME qml_load COMMAND alternativefastdelivery --smoke-test)
    set_tests_properties(qml_load PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1" TIMEOUT 20)
    add_test(NAME startup_interaction COMMAND alternativefastdelivery --interaction-smoke-test)
    set_tests_properties(startup_interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1" TIMEOUT 20)
  endif()
endif()
